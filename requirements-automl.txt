# Standalone environment for scripts/train_automl.py. Deliberately NOT layered
# on top of requirements.txt: flaml[automl] pins xgboost<3.0, which conflicts
# with this project's xgboost==3.2.0 (used by the XGBoost champion in
# train_final_model.py and by the Streamlit app). train_automl.py excludes
# "xgboost" from FLAML's search space, so the xgboost pulled in here as
# flaml's own transitive dependency is never actually used.
#
# Usage: python -m venv .venv-automl && source .venv-automl/bin/activate \
#        && pip install -r requirements-automl.txt \
#        && python scripts/train_automl.py
pandas==2.2.3
numpy==2.1.3
scikit-learn==1.6.1
scipy==1.15.3
joblib==1.4.2
flaml[automl]==2.6.0
